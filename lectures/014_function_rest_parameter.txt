Rest Parameter
    เป็นการทำให้ฟังก์ชันสามารถรับ parameter ได้ไม่จำกัด

    ตัวอย่างชุดคสั่ง
        function name( ...parameter: type[]): return_type {
            statement
        }

    ตัวอย่างการใช้งาน 
        function sum(...numbers: number[]): number {
            return numbers.reduce((previosVal: number, currentVal: number) => previosVal + currentVal);
        }
